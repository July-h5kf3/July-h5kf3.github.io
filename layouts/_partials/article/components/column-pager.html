{{/* Previous / next links within the post's column (系列文章); none for `syllabus.collection` columns (e.g. Blog). */}} {{- $cur := . -}} {{- with .GetTerms "column" -}} {{- $series := index . 0 -}} {{- $collection := false -}} {{- with $series.Params.syllabus }}{{ $collection = .collection }}{{ end -}} {{- $posts := cond $collection slice (partial "helper/column-pages" $series) -}} {{- $prev := "" -}}{{- $next := "" -}} {{- range $i, $p := $posts -}} {{- if eq $p.RelPermalink $cur.RelPermalink -}} {{- if gt $i 0 -}}{{- $prev = index $posts (sub $i 1) -}}{{- end -}} {{- if lt (add $i 1) (len $posts) -}}{{- $next = index $posts (add $i 1) -}}{{- end -}} {{- end -}} {{- end -}} {{- if or $prev $next -}}
{{- range $dir := slice "prev" "next" -}} {{- $p := cond (eq $dir "prev") $prev $next -}} {{- if $p -}} {{- $ct := partial "helper/column-title" $p -}} {{- if eq $dir "prev" }}{{ partial "helper/icon" "chevron-left" }}上一篇{{ else }}下一篇{{ partial "helper/icon" "chevron-right" }}{{ end -}} {{- if $ct.matched }}{{ $ct.track }}{{ end -}} {{ $ct.short }} {{- else -}} {{- end -}} {{- end -}}
