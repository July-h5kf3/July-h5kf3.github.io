{{/* Article-series navigation box. Renders on any post that belongs to a `column` taxonomy term ("系列文章"). Order comes from helper/column-pages (seriesOrder, else weight then date). Column-style titles ("[专栏][轨道] 短标题") are shown as a small 轨道 tag plus the short title; chapters announced in the column's `syllabus.upcoming` front matter are listed muted at the end. Columns with `syllabus.collection: true` (e.g. Blog) are loose collections, not ordered series: no box. */}} {{- $cur := . -}} {{- with .GetTerms "column" -}} {{- $series := index . 0 -}} {{- $collection := false -}} {{- with $series.Params.syllabus }}{{ $collection = .collection }}{{ end -}} {{- if not $collection -}} {{- $posts := partial "helper/column-pages" $series -}} {{- $upcoming := slice -}} {{- with $series.Params.syllabus -}}{{- $upcoming = .upcoming | default slice -}}{{- end -}} {{- $count := len $posts -}} {{- $pos := 0 -}} {{- range $i, $p := $posts -}}{{- if eq $p.RelPermalink $cur.RelPermalink -}}{{- $pos = add $i 1 -}}{{- end -}}{{- end -}}
{{ partial "helper/icon" "stack-2" }} 系列 {{ $series.LinkTitle }} {{ $pos }} / {{ $count }} {{ partial "helper/icon" "chevron-right" }}
    {{- range $i, $p := $posts -}} {{- $isCurrent := eq $p.RelPermalink $cur.RelPermalink -}} {{- $ct := partial "helper/column-title" $p -}}
  1. {{- if $isCurrent }} {{ add $i 1 }} {{- if $ct.matched }}{{ $ct.track }}{{ end -}} {{ $ct.short }} 正在阅读 {{- else }} {{ add $i 1 }} {{- if $ct.matched }}{{ $ct.track }}{{ end -}} {{ $ct.short }} {{- end }}
    2. {{- end -}} {{- range $j, $u := $upcoming -}} {{- $ct := partial "helper/column-title" (dict "Title" (printf "[%s][%s] %s" $series.LinkTitle $u.track $u.title)) -}}
  2. {{ add $count $j 1 }} {{ $ct.track }} {{ $ct.short }} 编写中
    3. {{- end -}}
{{- end -}} {{- end -}}
